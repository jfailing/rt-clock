//  DPR-AWARE CANVAS

//  Canvas backing store = CSS pixels × devicePixelRatio.

const NATIVE=920, CX=460, CY=460;
const canvas=document.getElementById('c');
const ctx=canvas.getContext('2d');

function applyDPR(cssPx){
  const dpr=window.devicePixelRatio||1;
  canvas.width=Math.round(cssPx*dpr);
  canvas.height=Math.round(cssPx*dpr);
  ctx.setTransform((cssPx*dpr)/NATIVE,0,0,(cssPx*dpr)/NATIVE,0,0);
}

// ═══════════════════════════════════════════════
//  SCALING
//  Clock ≈ 40 % of viewport width (capped by height).
//  Remaining width is the panel; digital clock fills it.
// ═══════════════════════════════════════════════
function rescale(){
  const vh=window.innerHeight, vw=window.innerWidth;
  // Clock: at most 40 % width or 92 % height, whichever is smaller
  const clockPx=Math.max(160,Math.min(Math.round(vw*0.40),Math.round(vh*0.92)));
  const wrap=document.getElementById('clockWrap');
  wrap.style.width=clockPx+'px'; wrap.style.height=clockPx+'px';
  applyDPR(clockPx);

  // Panel width available for the digital readout
  const panelPx=vw-clockPx-40;

  // Measure "00:00:00" at 100 px to find scale factor
  const PROBE=100;
  const el=document.createElement('span');
  el.style.cssText=`font-family:'Kalinga','Georgia',serif;font-size:${PROBE}px;position:fixed;visibility:hidden;white-space:nowrap`;
  el.textContent='00:00:00';
  document.body.appendChild(el);
  const pw=el.offsetWidth||520;
  document.body.removeChild(el);

  // Scale time to fill ~92 % of panel width; cap at 30 % of viewport height
  const timePx=Math.min(Math.round(panelPx*0.92/pw*PROBE),Math.round(vh*0.30));
  const datePx=Math.round(timePx*0.29);
  const bannerPx=Math.round(timePx*0.14);

  document.getElementById('timeEl').style.fontSize=timePx+'px';
  document.getElementById('dateEl').style.fontSize=datePx+'px';
  document.getElementById('banner').style.fontSize=bannerPx+'px';
}
window.addEventListener('resize',rescale);

function mTA(min){ return (min/60)*Math.PI*2-Math.PI/2; }

function getLayout(){
  const gap=6; let cursor=450; const layout=[];
  cfg.rings.forEach(ring=>{
    if(ring.radiusGroup){
      const leader=layout.find(l=>l.ring.id===ring.radiusGroup);
      if(leader){layout.push({ro:leader.ro,ri:leader.ri,ring,shared:true});return;}
    }
    const ro=cursor,ri=ro-ring.thickness;
    layout.push({ro,ri,ring,shared:false});
    cursor=ri-gap;
  });
  return layout;
}

function fillArc(ro,ri,s,e,color){
  ctx.save();ctx.fillStyle=color;ctx.beginPath();
  if(s===0&&e===60){
    ctx.arc(CX,CY,ro,0,Math.PI*2,false);ctx.arc(CX,CY,ri,0,Math.PI*2,true);
  }else{
    const a1=mTA(s),a2=mTA(e),top=-Math.PI/2;
    if(e>s){ctx.arc(CX,CY,ro,a1,a2,false);ctx.arc(CX,CY,ri,a2,a1,true);}
    else{ctx.arc(CX,CY,ro,a1,top+Math.PI*2,false);ctx.arc(CX,CY,ro,top,a2,false);ctx.arc(CX,CY,ri,a2,top,true);ctx.arc(CX,CY,ri,top+Math.PI*2,a1,true);}
  }
  ctx.closePath();ctx.fill();ctx.restore();
}

function radialTick(ro,ri,min,color,lw){
  const a=mTA(min);
  ctx.save();ctx.strokeStyle=color;ctx.lineWidth=lw||5;ctx.lineCap='round';
  ctx.beginPath();ctx.moveTo(CX+ro*Math.cos(a),CY+ro*Math.sin(a));ctx.lineTo(CX+ri*Math.cos(a),CY+ri*Math.sin(a));
  ctx.stroke();ctx.restore();
}

// T=0 marker — drawn in world coords, no ctx rotation
function drawT0(ro,ri,color,hw,shape,lr){
  lr=lr||1.0;
  const len=(ro-ri)*lr,riA=ro-len;
  const oy=CY-ro,iy=CY-riA,my=(oy+iy)/2;
  ctx.save();ctx.fillStyle=color;ctx.beginPath();
  switch(shape||'rectangle'){
    case 'rectangle': ctx.rect(CX-hw,oy,hw*2,len);break;
    case 'triangle':  ctx.moveTo(CX,oy);ctx.lineTo(CX-hw,iy);ctx.lineTo(CX+hw,iy);break;
    case 'diamond':   ctx.moveTo(CX,oy);ctx.lineTo(CX-hw,my);ctx.lineTo(CX,iy);ctx.lineTo(CX+hw,my);break;
    case 'trapezoid': ctx.moveTo(CX-hw,oy);ctx.lineTo(CX+hw,oy);ctx.lineTo(CX+hw*.5,iy);ctx.lineTo(CX-hw*.5,iy);break;
    case 'keystone':  ctx.moveTo(CX-hw*.5,oy);ctx.lineTo(CX+hw*.5,oy);ctx.lineTo(CX+hw,iy);ctx.lineTo(CX-hw,iy);break;
  }
  ctx.closePath();ctx.fill();ctx.restore();
}

function drawFace(faceR){
  const segs=cfg.face.segments||DC.face.segments;
  const showLbls=cfg.face.showQuadrantLabels===true;
  segs.forEach((seg,i)=>{
    const a1=mTA(i*15),a2=i===3?mTA(0)+Math.PI*2:mTA((i+1)*15);
    ctx.beginPath();ctx.moveTo(CX,CY);ctx.arc(CX,CY,faceR,a1,a2,false);ctx.closePath();ctx.fillStyle=seg.darkColor;ctx.fill();
    ctx.beginPath();ctx.moveTo(CX,CY);ctx.arc(CX,CY,faceR*.6,a1,a2,false);ctx.closePath();ctx.fillStyle=seg.lightColor;ctx.fill();
    ctx.save();ctx.strokeStyle='#000';ctx.lineWidth=2;ctx.beginPath();ctx.moveTo(CX,CY);ctx.lineTo(CX+faceR*Math.cos(a1),CY+faceR*Math.sin(a1));ctx.stroke();ctx.restore();
    if(showLbls&&seg.label){
      // Centred within quadrant (angular + radial midpoint)
      const midA=(a1+a2)/2, lr=faceR*.72;
      ctx.save();ctx.translate(CX+lr*Math.cos(midA),CY+lr*Math.sin(midA));
      ctx.font="bold 26px 'Kalinga','Trebuchet MS',sans-serif";ctx.fillStyle='rgba(255,255,255,.7)';ctx.textAlign='center';ctx.textBaseline='middle';
      ctx.fillText(seg.label,0,0);ctx.restore();
    }
  });
  drawTexture(faceR,cfg.face.texture||'none');
  if(cfg.face.showHourNumerals){
    for(let h=1;h<=12;h++){const a=mTA(h*5),nr=faceR*.86;ctx.save();ctx.translate(CX+nr*Math.cos(a),CY+nr*Math.sin(a));ctx.font="bold 18px 'Kalinga','Trebuchet MS',sans-serif";ctx.fillStyle='rgba(255,255,255,.6)';ctx.textAlign='center';ctx.textBaseline='middle';ctx.fillText(h,0,0);ctx.restore();}
  }
}

function drawTicks(innerR){

function drawClock(now){
  ctx.clearRect(0,0,NATIVE,NATIVE);
  const layout=getLayout();
  const own=layout.filter(l=>!l.shared);
  const faceR=own.length?own[own.length-1].ri-8:260;
  const outerR=own.length?own[0].ro:450;
  const g=cfg.global;

  // Rings
  layout.forEach(({ro,ri,ring,shared})=>{
    if(!shared&&(ring.startMin!==0||ring.endMin!==60))fillArc(ro,ri,0,60,'#060810');
    fillArc(ro,ri,ring.startMin,ring.endMin,ring.color);
    if(ring.over60&&ring.over60.enabled){const o=ring.over60,sr=ri+(ro-ri)*(1-o.subArcThicknessRatio);fillArc(ro,sr,o.subArcStart,o.subArcEnd,o.subArcColor);}
  });

  // Carets — anchored at ring's inner edge (ri), extend outward by caretLengthPx
  // or snap outer end to another ring's outer edge (ro) if caretSnapTo is set
  layout.forEach(({ri,ring})=>{
    if(ring.type!=='deadline'||ring.caretMin==null)return;
    const caretRi=ri;
    let caretRo;
    if(ring.caretSnapTo){
      const snap=layout.find(l=>l.ring.id===ring.caretSnapTo);
      caretRo=snap?snap.ro:ri+(ring.caretLengthPx||56);
    }else{
      caretRo=ri+(ring.caretLengthPx||56);
    }
    radialTick(caretRo,caretRi,ring.caretMin,ring.caretColor,5);
  });

  // T=0 marker

    const midMin=arcMid(ring.startMin,ring.endMin);
    ringLabel(ring.label,mid,midMin,ls);
  });

  drawTicks(own.length?own[own.length-1].ri:faceR+8);
  ctx.beginPath();ctx.arc(CX,CY,26,0,Math.PI*2);ctx.fillStyle='#040406';ctx.fill();

  // Hands

  const mA=mTA(m+s/60);
  const sA=secAngle(now);
  const sh=g.handShape||'baton';
  const mLen=g.minuteHandSnap?outerR:(g.minuteHandLength||420);

  ctx.beginPath();ctx.arc(CX,CY,13,0,Math.PI*2);ctx.fillStyle=g.secondHandColor||'#FFB800';ctx.fill();
  ctx.beginPath();ctx.arc(CX,CY, 5,0,Math.PI*2);ctx.fillStyle='#111';ctx.fill();
}

  }
  pX.restore();

// ── Hand drawing for preview ──

  pX.save();pX.translate(PCX,PCY);pX.rotate(angle);pX.shadowColor='rgba(0,0,0,.5)';pX.shadowBlur=3;
  const w=width,tail=8;
  switch(shape||'baton'){
    case 'baton': pX.beginPath();pX.moveTo(-tail,-w/2);pX.lineTo(len,-w/2);pX.lineTo(len,w/2);pX.lineTo(-tail,w/2);pX.closePath();pX.fillStyle=color;pX.fill();break;
    case 'triangle': pX.beginPath();pX.moveTo(len,0);pX.lineTo(-tail,-w/2);pX.lineTo(-tail,w/2);pX.closePath();pX.fillStyle=color;pX.fill();break;
    case 'sword': pX.beginPath();pX.moveTo(len,0);pX.lineTo(len*0.3,-w/2);pX.lineTo(-tail,-w*0.15);pX.lineTo(-tail,w*0.15);pX.lineTo(len*0.3,w/2);pX.closePath();pX.fillStyle=color;pX.fill();break;
    case 'skeleton': pX.beginPath();pX.moveTo(-tail,-w/2);pX.lineTo(len,-w/2);pX.lineTo(len,w/2);pX.lineTo(-tail,w/2);pX.closePath();pX.strokeStyle=color;pX.lineWidth=1.5;pX.stroke();break;
    case 'submariner':{const lr=w,tx=len;pX.beginPath();pX.moveTo(-tail,-w*0.42);pX.lineTo(tx-lr*1.1,-w*0.42);pX.lineTo(tx-lr*1.1,w*0.42);pX.lineTo(-tail,w*0.42);pX.closePath();pX.fillStyle=color;pX.fill();[0,120,240].forEach(d=>{const r=d*Math.PI/180;pX.beginPath();pX.arc(tx+Math.cos(r)*lr,Math.sin(r)*lr,lr,0,Math.PI*2);pX.fillStyle=color;pX.fill();});pX.beginPath();pX.arc(tx,0,lr*0.44,0,Math.PI*2);pX.fillStyle='rgba(0,0,0,.75)';pX.fill();break;}
    case 'twodoor':{const dr=w*1.4,dx=len-dr*0.9,sw=w*0.42;pX.beginPath();pX.moveTo(-tail,-sw*1.3);pX.lineTo(dx,-sw);pX.lineTo(dx,sw);pX.lineTo(-tail,sw*1.3);pX.closePath();pX.fillStyle=color;pX.fill();pX.beginPath();pX.arc(dx,0,dr,0,Math.PI*2);pX.fillStyle=color;pX.fill();pX.beginPath();pX.arc(dx,0,dr*0.38,0,Math.PI*2);pX.fillStyle='rgba(0,0,0,.65)';pX.fill();break;}

function pDrawSec(angle,len,color,lollipop){
  // Tip at (+len,0), tail at (-18,0)
  pX.save();pX.translate(PCX,PCY);pX.rotate(angle);pX.strokeStyle=color;pX.lineWidth=1.5;pX.lineCap='round';

function pSecAngle(now){
  const s=now.getSeconds(),ms=now.getMilliseconds(),st=cfg.global.secondStyle||'quartz';
  if(st==='springdrive')return mTA((s+ms/1000)/60*60);
  if(st==='mechanical')return mTA((s+Math.floor(ms/125)/8)/60*60);
  return mTA(s/60*60);
}

// ── Curved label ──
function drawCurvedTxt(text,r,midMin,sz,col){
  if(!text)return;const chars=text.split('');
  pX.font=`bold ${sz}px ${cfg.global.fontFamily},sans-serif`;
  const total=chars.reduce((s,c)=>s+pX.measureText(c).width/r,0);
  let ang=mTA(midMin)-total/2;
  pX.save();pX.fillStyle=col;
  chars.forEach(ch=>{pX.font=`bold ${sz}px ${cfg.global.fontFamily},sans-serif`;const w=pX.measureText(ch).width/r;const a=ang+w/2;pX.save();pX.translate(PCX+r*Math.cos(a),PCY+r*Math.sin(a));pX.rotate(a+Math.PI/2);pX.textAlign='center';pX.textBaseline='middle';pX.fillText(ch,0,0);pX.restore();ang+=w;});pX.restore();
}
function pLabel(text,r,midMin,ls){
  const glbl=cfg.labelStyle||DC.labelStyle;const inh=!ls||ls.inherit!==false;
  const curved=inh?glbl.curved:ls.curved,sz=inh?glbl.fontSize:ls.fontSize||12,col=inh?glbl.color:ls.color||'rgba(200,220,255,.75)';
  if(curved){drawCurvedTxt(text,r,midMin,sz,col);return;}
  const a=mTA(midMin);pX.save();pX.translate(PCX+r*Math.cos(a),PCY+r*Math.sin(a));pX.rotate(a+Math.PI/2);
  pX.font=`bold ${sz}px ${cfg.global.fontFamily},sans-serif`;pX.fillStyle=col;pX.textAlign='center';pX.textBaseline='middle';pX.fillText(text,0,0);pX.restore();
}

function renderPreview(){
  const now=new Date();
  const layout=getLayout();
  const own=layout.filter(l=>!l.shared);
  const faceR=own.length>0?own[own.length-1].ri-4:PMR*0.55;
  const outerR=own.length>0?own[0].ro:PMR;

  pX.clearRect(0,0,pC.width,pC.height);
  pX.fillStyle=cfg.global.bgColor;pX.fillRect(0,0,pC.width,pC.height);

  // rings
  layout.forEach(({ro,ri,ring,shared})=>{
    if(!shared&&(ring.startMin!==0||ring.endMin!==60))pFillArc(ro,ri,0,60,'#060810');
    pFillArc(ro,ri,ring.startMin,ring.endMin,ring.color);
    if(ring.over60&&ring.over60.enabled){const o=ring.over60,sr=ri+(ro-ri)*(1-o.subArcThicknessRatio);pFillArc(ro,sr,o.subArcStart,o.subArcEnd,o.subArcColor);}
  });

  // carets
  layout.forEach(({ro,ri,ring})=>{
    if(ring.type!=='deadline'||ring.caretMin==null)return;
    const lr=ring.caretLengthRatio!=null?ring.caretLengthRatio:1.0;
    let cRo,cRi;
    if(ring.radiusGroup){cRo=ro+14;cRi=ro-(ro-ri)*lr*0.4;}
    else{const ti=layout.findIndex(l=>l.ring.id===ring.caretBridgesTo);cRo=ti>=0?layout[ti].ri+1:ro+1;cRi=cRo-(cRo-(ri-1))*lr;}
    pTick(cRo,cRi,ring.caretMin,ring.caretColor,3);
  });

  // T0 — drawn at top of clock, no context rotation needed
  const t0=cfg.t0Caret||{color:'#CC1111',width:8,shape:'rectangle',lengthRatio:1.0};
  pDrawT0(outerR+2,faceR,t0.color,Math.max(2,t0.width*0.55),t0.shape||'rectangle',t0.lengthRatio||1.0);

  // ring labels
  layout.forEach(({ro,ri,ring,shared})=>{
    if(shared)return;
    const mid=(ro+ri)/2;const ls=ring.labelStyle||{inherit:true};
    const glbl=cfg.labelStyle||{};const inh=ls.inherit!==false;
    const centered=inh?(glbl.centered!==false):ls.centered!==false;
    const midMin=centered?arcMid(ring.startMin,ring.endMin):33;
    pLabel(ring.label,mid,midMin,ls);
  });

  // face segments
  cfg.face.segments.forEach((seg,i)=>{
    const a1=mTA(i*15),a2=i===3?mTA(0)+Math.PI*2:mTA((i+1)*15);
    pX.beginPath();pX.moveTo(PCX,PCY);pX.arc(PCX,PCY,faceR,a1,a2,false);pX.closePath();pX.fillStyle=seg.darkColor;pX.fill();
    pX.beginPath();pX.moveTo(PCX,PCY);pX.arc(PCX,PCY,faceR*0.6,a1,a2,false);pX.closePath();pX.fillStyle=seg.lightColor;pX.fill();
    pX.save();pX.strokeStyle='#000';pX.lineWidth=1;pX.beginPath();pX.moveTo(PCX,PCY);pX.lineTo(PCX+faceR*Math.cos(a1),PCY+faceR*Math.sin(a1));pX.stroke();pX.restore();
    if(cfg.face.showQuadrantLabels!==false){const ma=mTA(i*15+7.5),lr=faceR*0.76;pX.save();pX.translate(PCX+lr*Math.cos(ma),PCY+lr*Math.sin(ma));pX.font=`bold 12px ${cfg.global.fontFamily},sans-serif`;pX.fillStyle='rgba(255,255,255,.8)';pX.textAlign='center';pX.textBaseline='middle';pX.fillText(seg.label,0,0);pX.restore();}
  });
  drawTex(faceR,cfg.face.texture||'none');
  if(cfg.face.showHourNumerals){for(let h=1;h<=12;h++){const a=mTA(h*5),nr=faceR*0.86;pX.save();pX.translate(PCX+nr*Math.cos(a),PCY+nr*Math.sin(a));pX.font=`bold 11px ${cfg.global.fontFamily},sans-serif`;pX.fillStyle='rgba(255,255,255,.6)';pX.textAlign='center';pX.textBaseline='middle';pX.fillText(h,0,0);pX.restore();}}
  const tst=cfg.face.tickStyle||'simple';if(tst!=='none'){const tr=own.length>0?own[own.length-1].ri-2:faceR;for(let m=0;m<60;m++){const a=mTA(m),maj=m%5===0;const l2=tst==='railway'?(maj?12:m%2===0?7:4):(maj?10:5);pX.beginPath();pX.moveTo(PCX+tr*Math.cos(a),PCY+tr*Math.sin(a));pX.lineTo(PCX+(tr-l2)*Math.cos(a),PCY+(tr-l2)*Math.sin(a));pX.strokeStyle=maj?'rgba(255,255,255,.5)':'rgba(255,255,255,.18)';pX.lineWidth=maj?1.5:1;pX.stroke();}}
  pX.beginPath();pX.arc(PCX,PCY,14,0,Math.PI*2);pX.fillStyle='#040406';pX.fill();

  // hands
  const h=now.getHours()%12,m=now.getMinutes(),s=now.getSeconds();
  const hA=mTA(((h+m/60)/12)*60),mA=mTA(m+s/60),sA=pSecAngle(now);
  const g=cfg.global,sh=g.handShape||'baton';
  const mLen=g.minuteHandSnap?outerR:(g.minuteHandLength||420)*(PMR/450);
  const hLen=faceR*0.54,lollipop=sh==='submariner'||sh==='twodoor';
  pDrawHand(hA,hLen,g.hourHandWidth*0.65,g.hourHandColor||'#CCC',sh);
  pDrawMin(mA,mLen,g.minuteHandWidth*0.65,g.minuteHandColor||'#E0E0E0',sh);
  pDrawSec(sA,outerR*0.97,g.secondHandColor||'#FFB800',lollipop);
  pX.beginPath();pX.arc(PCX,PCY,7,0,Math.PI*2);pX.fillStyle=g.secondHandColor||'#FFB800';pX.fill();
  pX.beginPath();pX.arc(PCX,PCY,3,0,Math.PI*2);pX.fillStyle='#111';pX.fill();
}

    case 'guilloché': c.globalAlpha=0.09;for(let i=0;i<18;i++){const a=i*Math.PI/9;c.beginPath();c.moveTo(cx,cy);c.arc(cx,cy,r,a,a+Math.PI/9);c.closePath();c.fillStyle=i%2?'#000':'#fff';c.fill();}break;
    case 'sunburst':  c.globalAlpha=0.15;for(let i=0;i<60;i++){const a=(i/60)*Math.PI*2;c.beginPath();c.moveTo(cx,cy);c.lineTo(cx+r*Math.cos(a),cy+r*Math.sin(a));c.strokeStyle='rgba(255,255,255,.6)';c.lineWidth=0.6;c.stroke();}break;
    case 'fumé':      {const g=c.createRadialGradient(cx,cy,r*0.1,cx,cy,r);g.addColorStop(0,'rgba(180,200,255,.4)');g.addColorStop(1,'rgba(0,0,0,.7)');c.fillStyle=g;c.beginPath();c.arc(cx,cy,r,0,Math.PI*2);c.fill();}break;
    case 'sector':    c.globalAlpha=0.12;for(let i=0;i<8;i++){const a1=(i/8)*Math.PI*2,a2=((i+1)/8)*Math.PI*2;c.beginPath();c.moveTo(cx,cy);c.arc(cx,cy,r,a1,a2);c.closePath();c.fillStyle=i%2?'rgba(0,0,0,.15)':'rgba(255,255,255,.1)';c.fill();}break;
    case 'grand-seiko': c.globalAlpha=0.14;for(let x=cx-r;x<cx+r;x+=r*0.07){const dy=Math.sqrt(Math.max(0,r*r-(x-cx)**2));c.beginPath();c.moveTo(x,cy-dy);c.lineTo(x,cy+dy);c.strokeStyle='rgba(255,255,255,1)';c.lineWidth=0.5;c.stroke();}break;
    case 'regal-oak': {const sq=r*0.12;c.globalAlpha=0.11;for(let y=cy-r;y<cy+r;y+=sq)for(let x=cx-r;x<cx+r;x+=sq){if((x-cx)**2+(y-cy)**2>r*r)continue;c.fillStyle=(Math.round((x-cx)/sq)+Math.round((y-cy)/sq))%2===0?'rgba(255,255,255,.3)':'rgba(0,0,0,.2)';c.fillRect(x,y,sq,sq);}break;}
    case 'nemo':      {const sh=r*0.13;c.globalAlpha=0.15;for(let y=cy-r;y<cy+r;y+=sh){const dx=Math.sqrt(Math.max(0,r*r-(y-cy)**2));c.fillStyle=Math.round((y-(cy-r))/sh)%2===0?'rgba(255,255,255,.1)':'rgba(0,0,0,.1)';c.fillRect(cx-dx,y,dx*2,sh);}break;}
    case 'goldeneye': {c.globalAlpha=0.18;const wa=r*0.06,wl=r*0.28;for(let band=0;band<r*2/(r*0.1)+2;band++){const bY=cy-r+band*r*0.1;c.beginPath();let fi=true;for(let x=cx-r;x<=cx+r;x+=3){const y=bY+Math.sin((x/wl)*Math.PI*2)*wa;if(fi){c.moveTo(x,y);fi=false;}else c.lineTo(x,y);}c.strokeStyle=band%2===0?'rgba(100,160,255,.9)':'rgba(200,230,255,.6)';c.lineWidth=band%2===0?1:0.5;c.stroke();}break;}
  }
  c.restore();
}
